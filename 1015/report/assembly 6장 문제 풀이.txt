1. Filling an Array

; 6.11.2 - Exercise 1: Filling an Array
INCLUDE Irvine32.inc

.data
N = 10
array1 DWORD N DUP(?)
array2 DWORD N DUP(?)

.code
;-------------------------------------------------------------------
; FillArray: N개의 임의의 정수를 j...k 범위 내에서 생성하여 배열을 채움
; Receives: ESI = 배열 포인터, ECX = N, EBX = j (최소값), EDX = k (최대값)
; Returns: 없음 (모든 레지스터 값 보존)
;-------------------------------------------------------------------
FillArray PROC USES eax ebx ecx edx esi
    sub edx, ebx
    inc edx            ; EDX = 범위 크기 (k - j + 1)
L1:
    mov eax, edx
    call RandomRange   ; 0 ~ (k - j)
    add eax, ebx       ; j ~ k 범위로 보정
    mov [esi], eax
    add esi, TYPE DWORD
    loop L1
    ret
FillArray ENDP

main PROC
    call Randomize

    ; 테스트 1: j = -10, k = 10
    mov esi, OFFSET array1
    mov ecx, N
    mov ebx, -10
    mov edx, 10
    call FillArray

    ; 테스트 2: j = 100, k = 200
    mov esi, OFFSET array2
    mov ecx, N
    mov ebx, 100
    mov edx, 200
    call FillArray

    exit
main ENDP
END main

-------------------------------------------------------------------
; 6.11.2 - Exercise 2: Summing Array Elements in a Range
INCLUDE Irvine32.inc

.data
array SDWORD -5, 10, 20, 30, 45, -15, 60, 5, 0, 100
ArraySize = LENGTHOF array
sum1 DWORD ?
sum2 DWORD ?

.code
;-------------------------------------------------------------------
; SumRange: 배열 요소 중 j...k 범위에 속하는 값들의 합계를 반환
; Receives: ESI = 배열 포인터, ECX = 배열 크기, EBX = j, EDX = k
; Returns: EAX = 범위 내 요소들의 합계
;-------------------------------------------------------------------
SumRange PROC USES ebx ecx edx esi edi
    mov edi, 0         ; 합계 누적용
L1:
    mov eax, [esi]
    cmp eax, ebx       ; j 이상인지 검사
    jl Next
    cmp eax, edx       ; k 이하인지 검사
    jg Next
    add edi, eax       ; j <= element <= k 만족 시 가산
Next:
    add esi, TYPE SDWORD
    loop L1
    mov eax, edi       ; 합계를 EAX로 반환
    ret
SumRange ENDP

main PROC
    ; 테스트 1: 범위 0 ~ 50
    mov esi, OFFSET array
    mov ecx, ArraySize
    mov ebx, 0
    mov edx, 50
    call SumRange
    mov sum1, eax

    ; 테스트 2: 범위 -10 ~ 15
    mov esi, OFFSET array
    mov ecx, ArraySize
    mov ebx, -10
    mov edx, 15
    call SumRange
    mov sum2, eax

    exit
main ENDP
END main

; 6.11.2 - Exercise 4: College Registration
INCLUDE Irvine32.inc

.data
promptGrade   BYTE "Enter grade average (0 - 400): ", 0
promptCredits BYTE "Enter credits (1 - 30): ", 0
errMsgCredits BYTE "Error: Credits must be between 1 and 30.", 0dh, 0ah, 0
msgCanReg     BYTE "The student can register.", 0dh, 0ah, 0
msgCannotReg  BYTE "The student cannot register.", 0dh, 0ah, 0

gradeAverage  WORD ?
credits       WORD ?

.code
main PROC
    ; 성적 평점 입력
    mov edx, OFFSET promptGrade
    call WriteString
    call ReadInt
    mov gradeAverage, ax

    ; 학점 수 입력
    mov edx, OFFSET promptCredits
    call WriteString
    call ReadInt
    mov credits, ax

    ; 학점 수 범위 검사 (1 <= credits <= 30)
    cmp ax, 1
    jl InvalidCredits
    cmp ax, 30
    jg InvalidCredits
    jmp EvaluateRegistration

InvalidCredits:
    mov edx, OFFSET errMsgCredits
    call WriteString
    jmp Quit

EvaluateRegistration:
    ; CMP와 조건 분기만을 이용한 수강신청 자격 판별:
    ; 1) gradeAverage > 350 -> 가능
    ; 2) gradeAverage > 250 AND credits <= 16 -> 가능
    ; 3) credits <= 12 -> 가능
    mov ax, gradeAverage
    cmp ax, 350
    jg CanRegister

    cmp ax, 250
    jle CheckCreditsOnly
    mov bx, credits
    cmp bx, 16
    jle CanRegister

CheckCreditsOnly:
    mov bx, credits
    cmp bx, 12
    jle CanRegister

CannotRegister:
    mov edx, OFFSET msgCannotReg
    call WriteString
    jmp Quit

CanRegister:
    mov edx, OFFSET msgCanReg
    call WriteString

Quit:
    exit
main ENDP
END main

;-------------------------------------------------------------------
; EncryptMessage: 다중 문자 키(Multi-character key)를 이용한 XOR 암호화
; Receives: ESI = 메시지 버퍼, ECX = 메시지 길이, EDI = 키 버퍼, EDX = 키 길이
;-------------------------------------------------------------------
EncryptMessage PROC USES eax ebx ecx edx esi edi
    push edx           ; 키 길이 보존
    mov ebx, 0         ; 키 인덱스 초기화
L1:
    mov al, [edi + ebx]
    xor [esi], al
    inc esi
    inc ebx
    pop edx
    push edx
    cmp ebx, edx
    jl NextByte
    mov ebx, 0         ; 키 인덱스 순환
NextByte:
    loop L1
    pop edx
    ret
EncryptMessage ENDP

; Validate_PIN: 5자리 PIN 번호의 각 자릿수가 유효 범위 내인지 검사
; Receives: ESI = 5바이트 PIN 배열 포인터
; Returns: EAX = 유효하면 0, 오류 시 오류 위치 자릿수 (1..5)

.data

Validate_PIN PROC USES ebx ecx esi
    mov ecx, 5
    mov ebx, 0         ; 인덱스 0..4
CheckLoop:
    mov al, [esi + ebx]
    cmp al, minRange[ebx]
    jl ErrorFound
    cmp al, maxRange[ebx]
    jg ErrorFound
    inc ebx
    loop CheckLoop
    mov eax, 0         ; 성공
    ret
ErrorFound:
    mov eax, ebx
    inc eax            ; 1 기반 자릿수 위치 반환
    ret
Validate_PIN ENDP

;===================================================================
;-------------------------------------------------------------------
; CheckParity: 바이트 배열 전체의 비트 패리티 검사
; Receives: ESI = 배열 포인터, ECX = 배열 바이트 크기
; Returns: EAX = 짝수 패리티이면 1 (True), 홀수이면 0 (False)
;-------------------------------------------------------------------
CheckParity PROC USES ebx ecx esi
    mov bl, 0          ; 패리티 누적용
L1:
    xor bl, [esi]
    inc esi
    loop L1

    ; BL 패리티 검사
    mov al, bl
    xor al, 0          ; PF 플래그 설정
    jp EvenParity
    mov eax, 0         ; Odd Parity -> False
    ret
EvenParity:
    mov eax, 1         ; Even Parity -> True
    ret
CheckParity ENDP

    mov edx, OFFSET msgInvalid
    call WriteString
    call WriteDec
    call Crlf
DonePin:

    exit
main ENDP
END main
