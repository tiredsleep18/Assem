1. Display ASCII Decimal (WriteScaled)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

.data
DECIMAL_OFFSET = 5
decimal_one BYTE "100123456789765", 0

.code
WriteScaled PROC USES eax ebx ecx edx esi
    ; EDX = 문자열 오프셋, ECX = 문자열 길이, EBX = 소수점 위치
    mov esi, edx
    mov edx, ecx
    sub edx, ebx        ; 소수점이 삽입될 인덱스 계산

L1:
    cmp edx, 0
    jne PrintChar
    push eax
    mov al, '.'
    call WriteChar
    pop eax
PrintChar:
    mov al, [esi]
    call WriteChar
    inc esi
    dec edx
    loop L1
    ret
WriteScaled ENDP

2. Extended Subtraction Procedure (Extended_Sub)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

.code
Extended_Sub PROC USES eax ebx ecx edx esi edi
    ; ESI = 피감수 포인터, EDI = 감수 포인터, EBX = 결과 포인터, ECX = 바이트 크기
    clc                 ; Borrow(Carry) 초기화
L1:
    mov al, [esi]
    sbb al, [edi]       ; 빌림을 포함한 뺄셈
    mov [ebx], al
    inc esi
    inc edi
    inc ebx
    loop L1
    ret
Extended_Sub ENDP

3. Packed Decimal Conversion (PackedToAsc)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

.code
PackedToAsc PROC USES eax ebx ecx esi edi
    ; ESI = 4바이트 압축 BCD 포인터, EDI = ASCII 버퍼 포인터 (9바이트 이상)
    mov ecx, 4
    add esi, 3          ; Big-Endian 순서 처리를 위해 상위 바이트부터 진행
L1:
    mov al, [esi]
    mov bl, al
    shr al, 4           ; 상위 니블
    or  al, 30h         ; ASCII 변환
    mov [edi], al
    inc edi

    mov al, bl
    and al, 0Fh         ; 하위 니블
    or  al, 30h         ; ASCII 변환
    mov [edi], al
    inc edi

    dec esi
    loop L1
    mov BYTE PTR [edi], 0 ; Null 종단자
    ret
PackedToAsc ENDP

4. Greatest Common Divisor (GCD)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

.code
GCD PROC USES ebx edx
    ; Receives: EAX = x, EBX = y
    ; Returns:  EAX = gcd(x, y)
    cmp eax, 0
    jge L1
    neg eax             ; abs(x)
L1:
    cmp ebx, 0
    jge L2
    neg ebx             ; abs(y)
L2:
    cmp ebx, 0
    je Done

L_Loop:
    mov edx, 0
    div ebx             ; EDX = x % y, EAX = x / y
    mov eax, ebx        ; x = y
    mov ebx, edx        ; y = n
    cmp ebx, 0
    jg L_Loop

Done:
    ret
GCD ENDP

5. Bitwise Multiplication (BitwiseMultiply)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

.code
BitwiseMultiply PROC USES ebx ecx edx
    ; Receives: EAX = 피승수(multiplicand), EBX = 승수(multiplier)
    ; Returns:  EAX = 곱(product)
    mov edx, eax        ; EDX = 피승수
    mov eax, 0          ; EAX = 누적합 (0으로 초기화)

L1:
    cmp ebx, 0
    je Done
    shr ebx, 1          ; 승수의 LSB 검사
    jnc L2
    add eax, edx        ; LSB가 1이면 누적합에 추가
L2:
    shl edx, 1          ; 피승수 시프트
    jmp L1

Done:
    ret
BitwiseMultiply ENDP

6. Add Packed Decimal Integers (AddPacked)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

.code
AddPacked PROC USES eax ebx ecx edx esi edi
    ; ESI = 첫 번째 수, EDI = 두 번째 수, EDX = 합계 저장소, ECX = 바이트 크기
    clc                 ; Carry 초기화
L1:
    mov al, [esi]
    adc al, [edi]       ; Carry 포함 덧셈
    daa                 ; 압축 BCD 덧셈 보정
    mov [edx], al
    inc esi
    inc edi
    inc edx
    loop L1
    ret
AddPacked ENDP
