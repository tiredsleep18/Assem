[10.7 Review Questions and Exercises]

10.7.1 Short Answer
-------------------------------------------------------------------
1. What is the purpose of the STRUCT directive?
   - 답: 논리적으로 연관된 변수(필드)들을 하나의 데이터 구조체 템플릿으로 정의하여 묶어주는 지시어입니다.

2. Given RentalInvoice STRUCT invoiceNum BYTE 5 DUP(' '), dailyPrice WORD ?, daysRented WORD ? RentalInvoice ENDS:
   State whether or not each of the following declarations is valid:
   a. rentals RentalInvoice <>            -> Valid (기본 초기값으로 구조체 변수 선언)
   b. RentalInvoice rentals <>            -> Invalid (변수 이름과 구조체 타입의 순서가 바뀜)
   c. march RentalInvoice <'12345',10,0> -> Valid (필드 순서에 맞게 초기값 재정의)
   d. RentalInvoice <,10,0>               -> Invalid (변수 식별자 이름 누락)
   e. current RentalInvoice <,15,0,0>     -> Invalid (구조체 필드는 3개인데 초기값이 4개 전달됨)

3. (True/False): A macro cannot contain data definitions.
   - 답: False (매크로는 내부에서 .data 세그먼트를 열어 데이터 변수 정의를 포함할 수 있습니다).

4. What is the purpose of the LOCAL directive?
   - 답: 매크로 내부에서 선언된 라벨에 ??0000 형태의 고유 라벨명을 할당하여, 매크로가 여러 번 호출(확장)될 때 발생하는 중복 라벨 에러(Duplicate Label Error)를 방지합니다.

5. Which directive displays a message on the console during the assembly step?
   - 답: ECHO

6. Which directive marks the end of a conditional block of statements?
   - 답: ENDIF

7. List all the relational operators that can be used in constant boolean expressions.
   - 답: LT (Less Than), GT (Greater Than), EQ (Equal), NE (Not Equal), LE (Less Than or Equal), GE (Greater Than or Equal)

8. What is the purpose of the & operator in a macro definition?
   - 답: 치환 연산자(Substitution Operator)로, 매크로 파라미터가 큰따옴표 안의 문자열 리터럴이나 다른 텍스트와 붙어있을 때 해당 파라미터를 전달된 인자 값으로 강제 치환해 줍니다.

9. What is the purpose of the ! operator in a macro definition?
   - 답: 리터럴 문자 연산자(Literal-Character Operator)로, 특수 기호(<, >, %, ! 등)가 매크로 전처리기 기호로 해석되지 않고 단순 일반 문자로 취급되도록 강제합니다.

10. What is the purpose of the % operator in a macro definition?
    - 답: 확장 연산자(Expansion Operator)로, 텍스트 매크로를 확장하거나 매크로 호출 시 상수 수식을 계산된 텍스트 표현으로 변환하여 전달합니다.

4. 10개의 Triangle 구조체 배열을 선언하고 Vertex1을 (0..10, 0..10) 범위의 무작위 좌표로 초기화하는 루프:
   .data
   triangles Triangle 10 DUP(<>)
   .code
   mov ecx, 10
   mov edi, OFFSET triangles
   L1:
       mov eax, 11
       call RandomRange
       mov (Triangle PTR [edi]).Vertex1.X, ax
       mov eax, 11
       call RandomRange
       mov (Triangle PTR [edi]).Vertex1.Y, ax
       add edi, TYPE Triangle
       loop L1

       LOCAL L1
       push ecx
       push eax
       mov al, char
       mov ecx, count
   L1:
       call WriteChar
       loop L1
       pop eax
       pop ecx
   ENDM

1. mReadkey Macro
-------------------------------------------------------------------
INCLUDE Irvine32.inc

mReadkey MACRO asciiVar:REQ, scanVar:REQ
    call ReadChar
    mov asciiVar, al
    mov scanVar, ah
ENDM

.data
ascii BYTE ?
scan  BYTE ?

.code
main PROC
    mReadkey ascii, scan
    movzx eax, ascii
    call WriteHex
    call Crlf
    movzx eax, scan
    call WriteHex
    call Crlf
    exit
main ENDP
END main

5. mReadInt Macro

INCLUDE Irvine32.inc

mReadInt MACRO targetVar:REQ
    push eax
    call ReadInt
    IF (TYPE targetVar EQ 1)
        mov targetVar, al
    ELSEIF (TYPE targetVar EQ 2)
        mov targetVar, ax
    ELSEIF (TYPE targetVar EQ 4)
        mov targetVar, eax
    ELSE
        ECHO * Error: Invalid target variable size for mReadInt!
    ENDIF
    pop eax
ENDM

    mReadInt bVal
    mReadInt wVal
    mReadInt dVal
    exit

6. mWriteInt Macro
-------------------------------------------------------------------
INCLUDE Irvine32.inc

mWriteInt MACRO intVal:REQ
    push eax
    IF (TYPE intVal EQ 1)
        movsx eax, intVal
    ELSEIF (TYPE intVal EQ 2)
        movsx eax, intVal
    ELSEIF (TYPE intVal EQ 4)
        mov eax, intVal
    ELSE
        mov eax, intVal  ; 즉시값 또는 기본 레지스터 처리
    ENDIF
    call WriteInt
    pop eax
ENDM

.data
sByte SBYTE -15
sWord SWORD -1000
sDword SDWORD 50000

.code
main PROC
    mWriteInt sByte
    call Crlf
    mWriteInt sWord
    call Crlf
    mWriteInt sDword
    call Crlf
    exit
main ENDP
END main

7. The Professor's Lost Phone
-------------------------------------------------------------------
INCLUDE Irvine32.inc

WalkMax = 50
StartX = 25
StartY = 25

DrunkardWalk STRUCT
    path COORD WalkMax DUP(<0,0>)
    pathsUsed WORD 0
DrunkardWalk ENDS

.data
aWalk DrunkardWalk <>
dropStep WORD ?
phonePos COORD <0,0>
msgPhone BYTE "Lost Phone dropped at step ", 0
msgPos   BYTE " at coordinates (", 0
msgEnd   BYTE ")", 0dh, 0ah, 0

.code
main PROC
    call Randomize
    mov eax, WalkMax
    call RandomRange
    inc eax
    mov dropStep, ax

    mov esi, OFFSET aWalk
    mov edi, esi
    add edi, OFFSET DrunkardWalk.path
    mov ecx, WalkMax
    mov bx, StartX
    mov dx, StartY

    mov ax, 1
L1:
    mov (COORD PTR [edi]).X, bx
    mov (COORD PTR [edi]).Y, dx

    cmp ax, dropStep
    jne NextStep
    mov phonePos.X, bx
    mov phonePos.Y, dx

NextStep:
    push eax
    mov eax, 4
    call RandomRange
    cmp eax, 0
    je North
    cmp eax, 1
    je South
    cmp eax, 2
    je West
    inc bx
    jmp LoopCont
North:
    dec dx
    jmp LoopCont
South:
    inc dx
    jmp LoopCont
West:
    dec bx
LoopCont:
    pop eax
    add edi, TYPE COORD
    inc ax
    loop L1

    ; 핸드폰 떨어진 위치 출력
    mov edx, OFFSET msgPhone
    call WriteString
    movzx eax, dropStep
    call WriteDec
    mov edx, OFFSET msgPos
    call WriteString
    movzx eax, phonePos.X
    call WriteDec
    mov al, ','
    call WriteChar
    movzx eax, phonePos.Y
    call WriteDec
    mov edx, OFFSET msgEnd
    call WriteString

    exit
main ENDP
END main

8. Linked List
-------------------------------------------------------------------
INCLUDE Irvine32.inc

ListNode STRUCT
    NodeData DWORD ?
    NextPtr  DWORD ?
ListNode ENDS

TotalNodeCount = 5
NULL = 0
Counter = 0

.data
LinkedList LABEL PTR ListNode
REPEAT TotalNodeCount
    Counter = Counter + 1
    ListNode <Counter, ($ + Counter * SIZEOF ListNode)>
ENDM
ListNode <0, 0>

.code
main PROC
    mov esi, OFFSET LinkedList
NextNode:
    mov eax, (ListNode PTR [esi]).NextPtr
    cmp eax, NULL
    je Quit
    mov eax, (ListNode PTR [esi]).NodeData
    call WriteDec
    call Crlf
    mov esi, (ListNode PTR [esi]).NextPtr
    jmp NextNode
Quit:
    exit
main ENDP
END main

9. Shifting Multiple Doublewords (mShiftDoublewords)
-------------------------------------------------------------------
INCLUDE Irvine32.inc

mShiftDoublewords MACRO arrayName:REQ, direction:REQ, numberOfBits:REQ
    LOCAL L_Right, L_Left, Done
    push eax
    push ebx
    push ecx
    push esi

    mov bl, numberOfBits
    mov esi, OFFSET arrayName
    mov ecx, LENGTHOF arrayName - 1

    IFIDNI <direction>, <R>
        jmp L_Right
    ELSEIFIDNI <direction>, <L>
        jmp L_Left
    ENDIF

L_Right:
    push ecx
    mov eax, [esi + TYPE DWORD]
    mov cl, bl
    shrd [esi], eax, cl
    add esi, TYPE DWORD
    pop ecx
    loop L_Right
    shr DWORD PTR [esi], bl
    jmp Done

L_Left:
    add esi, (LENGTHOF arrayName - 1) * TYPE DWORD
L_LeftLoop:
    push ecx
    mov eax, [esi - TYPE DWORD]
    mov cl, bl
    shld [esi], eax, cl
    sub esi, TYPE DWORD
    pop ecx
    loop L_LeftLoop
    shl DWORD PTR [esi], bl

Done:
    pop esi
    pop ecx
    pop ebx
    pop eax
ENDM

.data
arrD DWORD 12345678h, 9ABCDEF0h, 11223344h

.code
main PROC
    mShiftDoublewords arrD, R, 4
    mShiftDoublewords arrD, L, 4
    exit
main ENDP
END main

10. Three-Operand Instructions
-------------------------------------------------------------------
INCLUDE Irvine32.inc

add3 MACRO dest:REQ, src1:REQ, src2:REQ
    mov eax, src1
    add eax, src2
    mov dest, eax
ENDM

sub3 MACRO dest:REQ, src1:REQ, src2:REQ
    mov eax, src1
    sub eax, src2
    mov dest, eax
ENDM

mul3 MACRO dest:REQ, src1:REQ, src2:REQ
    mov eax, src1
    imul eax, src2
    mov dest, eax
ENDM

div3 MACRO dest:REQ, src1:REQ, src2:REQ
    mov eax, src1
    cdq
    idiv src2
    mov dest, eax
ENDM

.data
w SDWORD 10
x SDWORD ?
y SDWORD 5
z SDWORD 3
temp SDWORD ?

.code
main PROC
    ; x = (w + y) * z 연산 수행
    add3 temp, w, y      ; temp = w + y (15)
    mul3 x, temp, z      ; x = temp * z (45)

    mov eax, x
    call WriteInt        ; +45 출력
    call Crlf
    exit
main ENDP
END main
